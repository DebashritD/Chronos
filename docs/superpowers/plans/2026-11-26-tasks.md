# Tasks Tab Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a Tasks tab (Today/Later to-dos that work offline and never use colour), and move habit management into Me.

**Architecture:**
- **API:** a new `com.chronos.task` slice (Flyway V3 table, JdbcClient repository, service, controller). Tasks use UUIDs made on the phone, and `PUT` can be sent twice without harm.
- **Web:** a `features/tasks/` slice. Its offline queue is modelled on `features/checkin/queue.ts`, using TanStack mutation defaults, persisted to IndexedDB, and in a separate scope from check-ins.
- **Navigation:** the tabs become Today · Tasks · Progress · Me. `/habits` stays as a route that isn't a tab, reached from Me.

**Tech Stack:** Spring Boot + JdbcClient + Flyway + Postgres (Testcontainers); React 19, react-router, TanStack Query v5 with persist-client, Tailwind v4, Vitest with Testing Library.

**Spec:** `docs/superpowers/specs/2026-11-26-tasks-design.md`. Mockups: `~/Docspace/Chronos/Todo Mockups/tasks-tab.html`.

## Global Constraints

**Rules**
- Tasks never use the mark colours (green, yellow or red classes, or `Mark`). Ticks are ink.
- Task text is trimmed and 1–200 characters. Dates must be no later than the user's today + 1.
- `GET /api/tasks` leaves out tasks with `done_on` before the user's today, so done tasks clear at midnight.
- Tab order is `/`, `/tasks`, `/progress`, `/settings`, labelled Today, Tasks, Progress and Me.

```java
package com.chronos.task;

import org.springframework.jdbc.core.simple.JdbcClient;
import org.springframework.stereotype.Repository;

import java.time.LocalDate;
import java.util.List;
import java.util.Optional;
import java.util.UUID;

@Repository
public class TaskRepository {

    private final JdbcClient jdbc;

    public TaskRepository(JdbcClient jdbc) {
        this.jdbc = jdbc;
    }

    /** Empty when the id belongs to another user: the conflict update is skipped and nothing is returned. */
    public Optional<Task> upsert(UUID id, long userId, String text, LocalDate scheduledOn, LocalDate doneOn) {
        return jdbc.sql("""
                        INSERT INTO task (id, user_id, text, scheduled_on, done_on)
                        VALUES (:id, :userId, :text, :scheduledOn, :doneOn)
                        ON CONFLICT (id) DO UPDATE
                          SET text = EXCLUDED.text, scheduled_on = EXCLUDED.scheduled_on, done_on = EXCLUDED.done_on
                          WHERE task.user_id = EXCLUDED.user_id
                        RETURNING *
                        """)
                .param("id", id).param("userId", userId).param("text", text)
                .param("scheduledOn", scheduledOn).param("doneOn", doneOn)
                .query(Task.class)
                .optional();
    }

    public List<Task> listFrom(long userId, LocalDate today) {
        return jdbc.sql("""
                        SELECT * FROM task
                        WHERE user_id = :userId AND (done_on IS NULL OR done_on >= :today)
                        ORDER BY created_at, id
                        """)
                .param("userId", userId).param("today", today)
                .query(Task.class)
                .list();
    }

    public int delete(UUID id, long userId) {
        return jdbc.sql("DELETE FROM task WHERE id = :id AND user_id = :userId").param("id", id).param("userId", userId).update();
    }

    public boolean exists(UUID id) {
        return jdbc.sql("SELECT EXISTS (SELECT 1 FROM task WHERE id = :id)").param("id", id).query(Boolean.class).single();
    }
}
```

```java
package com.chronos.task;

import com.chronos.core.error.ApiException;
import com.chronos.core.error.ErrorCode;
import com.chronos.core.user.UserService;
import org.springframework.stereotype.Service;

import java.time.LocalDate;
import java.util.List;
import java.util.UUID;

@Service
public class TaskService {

    private final TaskRepository tasks;
    private final UserService users;

    public TaskService(TaskRepository tasks, UserService users) {
        this.tasks = tasks;
        this.users = users;
    }

    public List<Task> list(long userId) {
        return tasks.listFrom(userId, users.today(userId));
    }

    public Task save(long userId, UUID id, TaskRequest body) {
        // One day of slack: a phone ahead of the server's idea of the user's today.
        LocalDate latest = users.today(userId).plusDays(1);
        if (isAfter(body.scheduledOn(), latest) || isAfter(body.doneOn(), latest)) {
            throw new ApiException(ErrorCode.VALIDATION, "Dates can’t be in the future");
        }
        return tasks.upsert(id, userId, body.text().strip(), body.scheduledOn(), body.doneOn())
                .orElseThrow(() -> ApiException.notFound("Task"));
    }

    public void delete(long userId, UUID id) {
        if (tasks.delete(id, userId) == 0 && tasks.exists(id)) {
            throw ApiException.notFound("Task");
        }
    }

    private static boolean isAfter(LocalDate date, LocalDate latest) {
        return date != null && date.isAfter(latest);
    }
}
```

```java
import com.chronos.core.user.AppUserPrincipal;
import jakarta.validation.Valid;
import org.springframework.http.HttpStatus;

import org.springframework.web.bind.annotation.RestController;

import java.util.List;
import java.util.UUID;

@RestController
@RequestMapping("/api/tasks")
public class TaskController {

    private final TaskService tasks;

    public TaskController(TaskService tasks) {
        this.tasks = tasks;
    }

    @GetMapping
    public List<TaskResponse> list(@AuthenticationPrincipal AppUserPrincipal principal) {
        return tasks.list(principal.id()).stream().map(TaskResponse::from).toList();
    }

    @PutMapping("/{id}")

    }

    @DeleteMapping("/{id}")
    @ResponseStatus(HttpStatus.NO_CONTENT)
    public void delete(@AuthenticationPrincipal AppUserPrincipal principal, @PathVariable UUID id) {
        tasks.delete(principal.id(), id);
    }
}
```

```ts
/** Runs a queued write; on a 401 it signs out like a query would, then waits for the same user and tries again. */
export async function authedSend<T>(fn: () => Promise<T>): Promise<T> {
  for (;;) {
    try {
      return await fn()
    } catch (error) {
      if (!(error instanceof ApiError && error.status === 401)) throw error
      notifyQueryError(error)
      await waitForAuth()
    }
  }
}
```

```ts
import { isCheckIn } from '../checkin/queue'
import { TASKS_KEY } from './api'

export const TASK_KEY = ['task'] as const
const MAX_SERVER_ERROR_RETRIES = 8

export type TaskVars = { op: 'put'; task: Task } | { op: 'delete'; id: string }

export function patchTasks(list: Task[], vars: TaskVars): Task[] {
  if (vars.op === 'delete') return list.filter((t) => t.id !== vars.id)
  return list.some((t) => t.id === vars.task.id)
    ? list.map((t) => (t.id === vars.task.id ? vars.task : t))
    : [...list, vars.task]
}

function send(vars: TaskVars): Promise<unknown> {
  if (vars.op === 'delete') return authedSend(() => api.del(`/api/tasks/${vars.id}`))
  const { id, text, scheduledOn, doneOn } = vars.task
  return authedSend(() => api.put(`/api/tasks/${id}`, { text, scheduledOn, doneOn }))
}

/** Refetch once the last queued change settles (the settling one still counts here). */
function synced(queryClient: QueryClient): Promise<void> | undefined {
  if (queryClient.isMutating({ mutationKey: TASK_KEY }) > 1) return undefined
  return queryClient.invalidateQueries({ queryKey: TASKS_KEY })
}

export function registerTaskDefaults(queryClient: QueryClient): void {
  queryClient.setMutationDefaults(TASK_KEY, {
    mutationFn: (vars: TaskVars) => send(vars),
    scope: { id: 'tasks' },
    retry: (failureCount, error) => {
      if (error instanceof AuthDiscardedError) return false
      if (!(error instanceof ApiError)) return true
      return error.status >= 500 && failureCount < MAX_SERVER_ERROR_RETRIES
    },
    onMutate: async (vars: TaskVars) => {
      await queryClient.cancelQueries({ queryKey: TASKS_KEY })
      queryClient.setQueryData<Task[]>(TASKS_KEY, (list) => patchTasks(list ?? [], vars))
    },
    onSuccess: () => synced(queryClient),
    onError: (error) => {
      if (error instanceof ApiError) {
        showSyncToast('Couldn’t sync a task. Refreshed.')
        return queryClient.invalidateQueries({ queryKey: TASKS_KEY })
      }
      return synced(queryClient)
    },
  })
}

export function useTaskChange(): (vars: TaskVars) => void {
  const { mutate } = useMutation<unknown, Error, TaskVars>({ mutationKey: TASK_KEY })
  return mutate
}

export function isTaskChange(mutation: Mutation): boolean {
  return mutation.options.mutationKey?.[0] === TASK_KEY[0]
}

/** What the sync line and the logout check count: marks and task changes still waiting. */
export function isQueuedChange(mutation: Mutation): boolean {
  return isCheckIn(mutation) || isTaskChange(mutation)
}

export async function resumeTasks(queryClient: QueryClient): Promise<void> {
  const pending = queryClient.getMutationCache().findAll({ mutationKey: TASK_KEY, status: 'pending' })
  await Promise.all(pending.map((m) => m.continue().catch(() => undefined)))
}

/** Call before discardCheckIns, which rejects the shared auth gate. */
export function discardTasks(queryClient: QueryClient): void {
  const cache = queryClient.getMutationCache()
  for (const m of cache.findAll({ mutationKey: TASK_KEY })) cache.remove(m)
}
```

```ts
import type { Task } from '../../api/types'
import { useNow } from '../../lib/useNow'
import { todayIn } from '../../lib/zonedTime'
import { browserTimezone, useSession } from '../auth/session'

const byCreated = (a: Task, b: Task) => a.createdAt.localeCompare(b.createdAt)

export function splitTasks(tasks: Task[], today: string) {
  const open = tasks.filter((t) => t.scheduledOn !== null && t.doneOn === null)
    .sort((a, b) => (a.scheduledOn ?? '').localeCompare(b.scheduledOn ?? '') || byCreated(a, b))
  const done = tasks.filter((t) => t.doneOn === today).sort(byCreated)
  const later = tasks.filter((t) => t.scheduledOn === null && t.doneOn === null).sort((a, b) => byCreated(b, a))
  return { open, done, later }
}

function daysBetween(from: string, to: string): number {
  return Math.round((Date.parse(`${to}T00:00:00Z`) - Date.parse(`${from}T00:00:00Z`)) / 86_400_000)
}

export function carriedText(scheduledOn: string, today: string): string | null {
  const days = daysBetween(scheduledOn, today)
  if (days <= 0) return null
  return days === 1 ? 'carried 1 day' : `carried ${days} days`
}

export function openTodayCount(tasks: Task[] | undefined, today: string): number {
  return tasks ? splitTasks(tasks, today).open.length : 0
}

export function useTaskToday(): string {
  const { me } = useSession()
  return todayIn(me?.timezone ?? browserTimezone(), useNow())
}
```

- [ ] **Step 2: Implement `useRowSwipe.ts`**

```ts
import { useRef, useState, type TouchEvent } from 'react'

const LATER_RATIO = 0.25
const DELETE_RATIO = 0.6
const LOCK_PX = 8

export type SwipeOutcome = 'none' | 'later' | 'delete'

export function swipeOutcome(dx: number, width: number, inLater: boolean): SwipeOutcome {
  if (dx >= 0 || width <= 0) return 'none'
  const ratio = -dx / width
  if (ratio > DELETE_RATIO || (inLater && ratio > LATER_RATIO)) return 'delete'
  return ratio > LATER_RATIO ? 'later' : 'none'
}

/** Leftward drag on one row; vertical drags are left to the page scroll. */
export function useRowSwipe(inLater: boolean, onCommit: (outcome: Exclude<SwipeOutcome, 'none'>) => void) {
  const [dx, setDx] = useState(0)
  const start = useRef<{ x: number; y: number; width: number; lock: 'x' | 'y' | null } | null>(null)

  return {
    dx,
    outcome: swipeOutcome(dx, start.current?.width ?? 0, inLater),
    handlers: {
      onTouchStart(e: TouchEvent<HTMLElement>) {
        const t = e.touches[0]
        start.current = { x: t.clientX, y: t.clientY, width: e.currentTarget.getBoundingClientRect().width, lock: null }
      },
      onTouchMove(e: TouchEvent<HTMLElement>) {
        const s = start.current
        if (!s) return
        const t = e.touches[0]
        const mx = t.clientX - s.x, my = t.clientY - s.y
        if (s.lock === null && Math.max(Math.abs(mx), Math.abs(my)) > LOCK_PX) s.lock = Math.abs(mx) > Math.abs(my) ? 'x' : 'y'
        if (s.lock === 'x') setDx(Math.min(0, mx))
      },
      onTouchEnd() {
        const s = start.current
        const outcome = s ? swipeOutcome(dx, s.width, inLater) : 'none'
        start.current = null
        setDx(0)
        if (outcome !== 'none') onCommit(outcome)
      },
    },
  }
}
```

`onTouchEnd` reads `dx` from the closure. If the test shows it's stale, keep the latest dx in a ref, set it in `onTouchMove`, and read it in `onTouchEnd`.
